Examples#

Use the single-rate example when one periodic runnable is sufficient for a function, for example simple signal conditioning or threshold checks with one execution rhythm.

Use the multi-rate example when preprocessing and decision logic should run at different cycle times, for example 10 ms input conditioning and 20 ms output publication.

 41FunctionResult SpeedHazardDetection::init(
 42    SpeedHazardDetectionContext& context,

 45    // Validate parameters

 54    // Reset context state

 56    
 57    // Initialize with safe defaults
 58    context.previousSpeed = 0.0f;

 60    context.hazardRequestActive = false;
 61    context.hazardRequestStartTimestamp = 0;
 62    context.cycleCounter = 0;
 63    
 64    // Mark as initialized
 65    context.isInitialized = true;
 66    
 67    return FunctionResult::OK;
 68}

 70FunctionResult SpeedHazardDetection::step(
 71    SpeedHazardDetectionContext& context,
 72    const SpeedHazardDetectionParameters& parameters,
 73    const SpeedHazardDetectionInputs& inputs,
 74    SpeedHazardDetectionOutputs& outputs) {
 75    
 76    // Check initialization
 77    if (!context.isInitialized) {
 78        outputs.functionStatus = FunctionResult::NOT_AVAILABLE;
 79        outputs.hazardRequest.value = false;
 80        outputs.hazardRequest.quality = DataQuality::INVALID;
 81        return FunctionResult::NOT_AVAILABLE;
 82    }
 83    
 84    // Increment cycle counter
 85    context.cycleCounter++;
 86    
 87    // ========================================================================
 88    // INPUT VALIDATION
 89    // ========================================================================
 90    
 91    // Check vehicle speed data quality
 92    if (inputs.vehicleSpeed.quality == DataQuality::INVALID ||
 93        inputs.vehicleSpeed.quality == DataQuality::UNINITIALIZED) {
 94        outputs.functionStatus = FunctionResult::INVALID_INPUT;
 95        outputs.hazardRequest.quality = DataQuality::INVALID;
 96        return FunctionResult::INVALID_INPUT;
 97    }
 98    
 99    // Check acceleration data quality
100    if (inputs.accelerationLongitudinal.quality == DataQuality::INVALID ||
101        inputs.accelerationLongitudinal.quality == DataQuality::UNINITIALIZED) {
102        outputs.functionStatus = FunctionResult::INVALID_INPUT;
103        outputs.hazardRequest.quality = DataQuality::INVALID;
104        return FunctionResult::INVALID_INPUT;
105    }
106    
107    // Validate speed range
108    if (inputs.vehicleSpeed.value < 0.0f || inputs.vehicleSpeed.value > 300.0f) {
109        outputs.functionStatus = FunctionResult::RANGE_ERROR;
110        outputs.hazardRequest.quality = DataQuality::INVALID;
111        return FunctionResult::RANGE_ERROR;
112    }
113    
114    // Validate acceleration range
115    if (inputs.accelerationLongitudinal.value < -20.0f ||
116        inputs.accelerationLongitudinal.value > 20.0f) {
117        outputs.functionStatus = FunctionResult::RANGE_ERROR;
118        outputs.hazardRequest.quality = DataQuality::INVALID;
119        return FunctionResult::RANGE_ERROR;
120    }
121    
122    // ========================================================================
123    // SPEED INCREASE DETECTION LOGIC
124    // ========================================================================
125    
126    float currentSpeed = inputs.vehicleSpeed.value;
127    float speedIncrease = 0.0f;
128    
129    if (context.previousSpeedTimestamp > 0) {
130        // Calculate time delta in seconds
131        uint32_t timeDeltaMs = inputs.vehicleSpeed.updateTimestampMs - 
132                               context.previousSpeedTimestamp;
133        float timeDeltaS = timeDeltaMs / 1000.0f;
134        
135        // Ensure we have a valid time delta (at least 1ms)
136        if (timeDeltaS > 0.0f) {
137            speedIncrease = currentSpeed - context.previousSpeed;
138        }
139    }
140    
141    // Calculate relative speed increase in percent
142    float relativeSpeedIncreasePercent = 0.0f;
143    if (context.previousSpeed > 0.1f) {  // Avoid division by zero
144        relativeSpeedIncreasePercent = 
145            (speedIncrease / context.previousSpeed) * 100.0f;
146    }
147    
148    // ========================================================================
149    // HAZARD REQUEST DECISION LOGIC
150    // ========================================================================
151    
152    bool shouldActivateHazard = false;
153    
154    // Check if acceleration threshold triggers hazard for forward motion
155    if (currentSpeed > 5.0f &&  // Only when vehicle is actively moving
156        inputs.accelerationLongitudinal.value > 0.0f &&  // Forward acceleration
157        relativeSpeedIncreasePercent >= parameters.speedHazardForwardThresholdPercent) {
158        shouldActivateHazard = true;
159    }
160    
161    // ========================================================================
162    // HAZARD REQUEST LATCHING
163    // ========================================================================
164    
165    if (shouldActivateHazard) {
166        // Activate or re-trigger the hazard request
167        context.hazardRequestActive = true;
168        context.hazardRequestStartTimestamp = inputs.vehicleSpeed.updateTimestampMs;
169    }
170    
171    // Check if hazard request should remain active based on duration
172    if (context.hazardRequestActive) {
173        uint32_t hazardDurationElapsedMs = 
174            inputs.vehicleSpeed.updateTimestampMs - context.hazardRequestStartTimestamp;
175        
176        if (hazardDurationElapsedMs >= parameters.hazardRequestDurationMs) {
177            // Hazard request duration has expired
178            context.hazardRequestActive = false;
179        }
180    }
181    
182    // ========================================================================
183    // OUTPUT ASSIGNMENT
184    // ========================================================================
185    
186    outputs.hazardRequest.value = context.hazardRequestActive;
187    outputs.hazardRequest.quality = DataQuality::VALID;
188    outputs.hazardRequest.updateTimestampMs = inputs.vehicleSpeed.updateTimestampMs;
189    outputs.functionStatus = FunctionResult::OK;
190    
191    // ========================================================================
192    // STATE UPDATE FOR NEXT CYCLE
193    // ========================================================================
194    
195    context.previousSpeed = currentSpeed;
196    context.previousSpeedTimestamp = inputs.vehicleSpeed.updateTimestampMs;
197    
198    return FunctionResult::OK;
199}

201FunctionResult SpeedHazardDetection::terminate(
202    SpeedHazardDetectionContext& context,
203    SpeedHazardDetectionOutputs& outputs) {
204    
205    // Ensure all outputs are in safe state
206    outputs.hazardRequest.value = false;
207    outputs.hazardRequest.quality = DataQuality::VALID;
208    outputs.hazardRequest.updateTimestampMs = 0;
209    outputs.functionStatus = FunctionResult::OK;
210    
211    // Reset internal state
212    context.reset();
213    
214    return FunctionResult::OK;
215}
